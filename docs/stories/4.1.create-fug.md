class CreateEventScreen extends ConsumerStatefulWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Form(
        child: Stepper(
          type: StepperType.vertical,
          currentStep: state.currentStep,
          steps: [
            // Step 1: Information (title, description, image, type)
            Step(
              title: Text(l10n.information),
              content: Column(/* form fields */),
            ),
            // Step 2: Date and Time
            Step(
              title: Text(l10n.dateAndTime),
              content: Column(/* date pickers */),
            ),
            // Step 3: Location
            Step(
              title: Text(l10n.location),
              content: Column(/* map picker */),
            ),
            // Step 4: Options
            Step(
              title: Text(l10n.options),
              content: Column(/* max participants, tags */),
            ),
          ],
        ),
      ),
    );
  }
}
