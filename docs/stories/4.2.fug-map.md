class EventsMapScreen extends ConsumerStatefulWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Stack(
        children: [
          FlutterMap(
            mapController: _mapController,
            options: MapOptions(
              initialCenter: _mapCenter,
              initialZoom: 13,
              onMapEvent: (event) {
                if (event is MapEventMoveEnd) {
                  _loadNearbyEvents();
                }
              },
            ),
            children: [
              TileLayer(
                urlTemplate: isDarkMode
                    ? 'https://basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png'
                    : 'https://tile.openstreetmap.org/{z}/{x}/{y}.png',
              ),
              // User position circle
              if (_userGpsPosition != null)
                CircleLayer(circles: [/* accuracy circle */]),
              // User marker
              if (_userGpsPosition != null)
                MarkerLayer(markers: [/* user marker */]),
              // Events cluster
              MarkerClusterLayerWidget(
                options: MarkerClusterLayerOptions(
                  markers: _markers,
                  builder: (context, markers) => /* cluster widget */,
                ),
              ),
            ],
          ),
          // Search radius indicator
          // Loading indicator
          // Error message
        ],
      ),
    );
  }
}
